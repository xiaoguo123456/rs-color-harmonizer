cmake_minimum_required(VERSION 3.20)
project(rs-color-harmonizer VERSION 0.1.0 LANGUAGES CXX)

option(RSCH_BUILD_TESTS "Build synthetic processing tests" ON)
option(RSCH_BUILD_CLI "Build the rsch command-line utility" ON)
option(RSCH_BUILD_QT "Build the optional Qt Quick shell" OFF)
option(RSCH_USE_GDAL "Use GDAL when available (the built-in TIFF reader remains available)" ON)
option(RSCH_USE_OPENCV "Enable OpenCV acceleration hooks when available" ON)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

add_library(rsch_core STATIC src/core.cpp)
target_include_directories(rsch_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(rsch_core PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(rsch_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(rsch_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()

if(RSCH_USE_GDAL)
  find_package(GDAL QUIET)
  if(GDAL_FOUND)
    message(STATUS "GDAL found: ${GDAL_VERSION}; core keeps a dependency-free TIFF fallback")
    target_compile_definitions(rsch_core PUBLIC RSCH_WITH_GDAL=1)
    if(TARGET GDAL::GDAL)
      target_link_libraries(rsch_core PUBLIC GDAL::GDAL)
    else()
      target_include_directories(rsch_core PUBLIC ${GDAL_INCLUDE_DIRS} ${GDAL_INCLUDE_DIR})
      if(GDAL_LIBRARIES)
        target_link_libraries(rsch_core PUBLIC ${GDAL_LIBRARIES})
      else()
        target_link_libraries(rsch_core PUBLIC ${GDAL_LIBRARY})
      endif()
    endif()
  else()
    message(STATUS "GDAL development files not found; using built-in GeoTIFF reader")
  endif()
endif()
if(RSCH_USE_OPENCV)
  find_package(OpenCV QUIET COMPONENTS core imgproc)
  if(OpenCV_FOUND)
    message(STATUS "OpenCV found: ${OpenCV_VERSION}")
    target_compile_definitions(rsch_core PUBLIC RSCH_WITH_OPENCV=1)
    target_link_libraries(rsch_core PUBLIC ${OpenCV_LIBS})
  else()
    message(STATUS "OpenCV not found; using standard-library processing kernels")
  endif()
endif()

if(RSCH_BUILD_CLI)
  add_executable(rsch apps/rsch.cpp)
  target_link_libraries(rsch PRIVATE rsch_core)
endif()

if(RSCH_BUILD_TESTS)
  enable_testing()
  add_executable(rsch_core_tests tests/test_core.cpp)
  target_link_libraries(rsch_core_tests PRIVATE rsch_core)
  add_test(NAME rsch_core_tests COMMAND rsch_core_tests)
endif()

if(RSCH_BUILD_QT)
  find_package(Qt6 QUIET COMPONENTS Core Quick Qml)
  if(Qt6_FOUND)
    qt_add_executable(rsch_gui apps/rsch_gui.cpp)
    qt_add_qml_module(rsch_gui URI RSColorHarmonizer VERSION 1.0 QML_FILES qml/Main.qml)
    target_link_libraries(rsch_gui PRIVATE rsch_core Qt6::Core Qt6::Quick Qt6::Qml)
  else()
    message(WARNING "Qt6 not found; RSCH_BUILD_QT requested but GUI is disabled")
  endif()
endif()

include(GNUInstallDirs)
install(TARGETS rsch_core ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
if(RSCH_BUILD_CLI)
  install(TARGETS rsch RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES qml/Main.qml DESTINATION ${CMAKE_INSTALL_DATADIR}/rs-color-harmonizer/qml)
